local SECRET = 'kh3n_s3cr3t_x0r_k3y_2024';   -- Voltei para a antiga

local function computeSig(id)
    local out = {};
    for i = 1, #id do
        local b = bit32.bxor(id:byte(i), SECRET:byte((i - 1) % #SECRET + 1));
        out[#out + 1] = string.format('%02x', b);
    end;
    return table.concat(out);
end;
local B87 = '.CDJMHOfYQIPkL8X2y$h!gbqUaj6di9Rl(wm0sur4<vpoz#[nt;1S)e>&^c?KVE-A~_/@:*xW3+5=|7%NTG]FBZ';
local function decodeBase87(s)
    local out = {};
    local i = 1;
    while i <= #s do
        local chunk = s:sub(i, i + 4);
        local pad = 5 - #chunk;
        local val = 0;
        for j = 1, #chunk do
            local c = B87:find(chunk:sub(j, j), 1, true);
            if not c then
                error('invalid.'); 
            end;
            val = val * 87 + c - 1; 
        end;
        val = val * 87 ^ pad;
        for k = 3, 0, -1 do
            if k < 4 - pad then
                table.insert(out, string.char(math.floor(val / 256 ^ k) % 256)); 
            end; 
        end;
        i = i + 5; 
    end;
    return table.concat(out); 
end;
local function notifyError(reasonMsg)
    game:GetService('StarterGui'):SetCore('SendNotification', {
        Title = 'script',
        Text = reasonMsg,
        Duration = 5
    });
end

local REASON_MESSAGES = {
    deleted = 'this script has been deleted or is no longer valid.',
    invalid_sig = 'verification failed. please rejoin and try again.',
    token_expired = 'session expired. please run the script again.',
    rate_limited = 'too many requests. please wait a moment and try again.',
    invalid_request = 'invalid request. please run the script again.',
    server_error = 'server error. please try again in a moment.'
}

local function getErrorMessage(res)
    if not res then
        return 'could not connect to the server. check your connection and try again.'
    end
    local ok, body = pcall(function() return HttpService:JSONDecode(res.Body) end)
    if ok and body and body.reason and REASON_MESSAGES[body.reason] then
        return REASON_MESSAGES[body.reason]
    end
    return 'this script has been deleted or is no longer valid.'
end

local sig = computeSig(genv.script_id);
local tokenRes;
pcall(function()
    tokenRes = request({
        Url = 'https://protectscripts.vercel.app/api/auth?script_id=' .. genv.script_id,
        Method = 'GET',
        Headers = {
            ['Accept'] = 'application/json',
            ['X-Sig'] = sig
        }
    }); 
end);
if not tokenRes or not tokenRes.Body or tokenRes.StatusCode ~= 200 then
    notifyError(getErrorMessage(tokenRes));
    _warn("xd token: " .. tostring(tokenRes and tokenRes.StatusCode or "nil"))
    return
end

local ok2, parsed = pcall(function() return HttpService:JSONDecode(tokenRes.Body) end)
if not ok2 or not parsed or not parsed.token then

end
